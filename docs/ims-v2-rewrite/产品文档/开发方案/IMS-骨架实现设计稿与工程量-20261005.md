## 2. `cmd/api/main.go` 骨架

> 装配顺序（冻结）：**配置加载 → 双 DB + Redis → 中间件链 → 路由注册 → 优雅关闭**。

```go
// cmd/api/main.go
package main

import (
	"context"
	"errors"
	"net/http"
	"os"
	"os/signal"
	"syscall"
	"time"

	"go.uber.org/zap"

	"github.com/shenyu/ims/internal/bootstrap"
	"github.com/shenyu/ims/internal/config"
)

func main() {
	// 1) 配置加载（viper：文件 + 环境变量；键见 §8）
	cfg, err := config.Load()
	if err != nil {
		panic("config load failed: " + err.Error())
	}

	// 2) 日志（zap，结构化；环境区分 JSON / Console）
	logger := config.NewLogger(cfg.App.Env)
	defer func() { _ = logger.Sync() }()

	// 3) 双 DB（OPSDB 同库直连 oa_* + IMSDB 自管表）+ Redis
	runtime, err := bootstrap.NewRuntime(cfg, logger)
	if err != nil {
		logger.Fatal("bootstrap runtime failed", zap.Error(err))
	}
	defer runtime.Close()

	// 4) 中间件链装配 + 路由注册
	router := bootstrap.NewRouter(cfg, runtime, logger)

	srv := &http.Server{
		Addr:              cfg.App.HTTPAddr,          // 默认 :8080
		Handler:           router,
		ReadHeaderTimeout: 5 * time.Second,
		ReadTimeout:       15 * time.Second,
		WriteTimeout:      30 * time.Second,
		IdleTimeout:       60 * time.Second,
	}

	// 5) 启动 + 优雅关闭
	go func() {
		logger.Info("ims-api listening", zap.String("addr", cfg.App.HTTPAddr))
		if err := srv.ListenAndServe(); err != nil && !errors.Is(err, http.ErrServerClosed) {
			logger.Fatal("listen failed", zap.Error(err))
		}
	}()

	quit := make(chan os.Signal, 1)
	signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)
	<-quit
	logger.Info("shutdown signal received, draining...")

	ctx, cancel := context.WithTimeout(context.Background(), 20*time.Second)
	defer cancel()
	if err := srv.Shutdown(ctx); err != nil {
		logger.Error("graceful shutdown failed", zap.Error(err))
	}
	logger.Info("ims-api stopped")
}
```

### 2.1 `internal/bootstrap/router.go` —— 中间件链装配顺序（冻结）

```go
// internal/bootstrap/router.go
package bootstrap

import (
	"net/http"

	"github.com/go-chi/chi/v5"
	"github.com/go-chi/chi/v5/middleware"
	"go.uber.org/zap"

	"github.com/shenyu/ims/internal/authmw"   // 见 §5（中间件目录；命名不叫 middleware 以避免与 chi 冲突）
	"github.com/shenyu/ims/internal/config"
)

// 网关前缀：/admin-api/ims/（ADR-IMS-002；前端 Nginx 反代同前缀）
const apiPrefix = "/admin-api/ims"

func NewRouter(cfg *config.Config, rt *Runtime, logger *zap.Logger) http.Handler {
	r := chi.NewRouter()

	// ── 全局横切（不含鉴权）──
	r.Use(authmw.Recover(logger))                       // 1. panic 恢复 → CommonResult 5001
	r.Use(middleware.RequestID)                          // X-Request-Id（日志串联）
	r.Use(authmw.RequestLog(logger))                     // 2. 访问日志（method/path/cost/uid/traceId）
	r.Use(authmw.CORS(cfg.CORS))                         // 3. CORS（白名单来源）
	r.Use(middleware.RealIP)

	// ── 健康检查（免鉴权、免 B9）──
	r.Get(apiPrefix+"/health", func(w http.ResponseWriter, _ *http.Request) {
		_, _ = w.Write([]byte(`{"code":0,"msg":"success","data":{"status":"UP"}}`))
	})

	// ── 公开端点（钉钉 SSO / 事件回调；不走 B9 数据范围）──
	r.Group(func(pub chi.Router) {
		pub.Use(authmw.BodyLimit(cfg.App.MaxBodyBytes))
		mountPublicRoutes(pub, cfg, rt) // /auth/sso/redirect、/auth/sso/callback、/callback/dingtalk/*
	})

	// ── 受保护端点（JWT → TenantInject → RowScope(B9)）──
	r.Group(func(prot chi.Router) {
		prot.Use(authmw.JWTAuth(cfg.JWT, rt.Redis))          // 4. 解析 JWT → 构建 LoginContext
		prot.Use(authmw.TenantInject())                      // 5. 注入 ctx.tenantId（V1 恒 0，请求体不可覆盖）
		prot.Use(authmw.RowScope(rt.Resolver, rt.Registry))  // 6. B9：dataScope/BR-212/fail-closed/1504

		mountAuthRoutes(prot, rt)   // /auth/**（AUTH-002/003/004，除 SSO）
		mountSysRoutes(prot, rt)    // /system/**
	})

	return r
}
```

### 2.2 `internal/bootstrap/db.go` —— 双 DB + sqlc 双 Queries

```go
// internal/bootstrap/db.go
package bootstrap

import (
	"database/sql"
	"time"

	_ "github.com/go-sql-driver/mysql"

	"github.com/shenyu/ims/internal/config"
	"github.com/shenyu/ims/internal/db/sqlcgen/imsdb"
	"github.com/shenyu/ims/internal/db/sqlcgen/opsdb"
)

// OPSDB：同库直连现网 oa_*（只读为主；只加列兼容 IR-04）
// IMSDB：IMS 自管表（ims_sys_* / ims_auth_* / ims_role_*）
type DBs struct {
	OPS *sql.DB
	IMS *sql.DB

	OpsQ *opsdb.Queries // sqlc 生成：对 db/query/ops/**
	ImsQ *imsdb.Queries // sqlc 生成：对 db/query/ims/**
}

func openDB(dsn string, maxOpen, maxIdle int) (*sql.DB, error) {
	db, err := sql.Open("mysql", dsn)
	if err != nil {
		return nil, err
	}
	db.SetMaxOpenConns(maxOpen)
	db.SetMaxIdleConns(maxIdle)
	db.SetConnMaxLifetime(30 * time.Minute)
	if err := db.Ping(); err != nil {
		return nil, err
	}
	return db, nil
}

func NewDBs(cfg *config.Config) (*DBs, error) {
	ops, err := openDB(cfg.DB.OPSDSN, cfg.DB.OPSMaxOpen, cfg.DB.OPSMaxIdle)
	if err != nil {
		return nil, err
	}
	ims, err := openDB(cfg.DB.IMSDSN, cfg.DB.IMSMaxOpen, cfg.DB.IMSMaxIdle)
	if err != nil {
		return nil, err
	}
	return &DBs{
		OPS: ops, IMS: ims,
		OpsQ: opsdb.New(ops),
		ImsQ: imsdb.New(ims),
	}, nil
}
```

```go
		logger.Fatal("bootstrap runtime failed", zap.Error(err))
	}
	defer rt.Close()

	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()

	// 1) outbox 消费者（常驻；拉取 PENDING 事件 → 幂等 upsert → 幂等归还）
	consumer := worker.NewOutboxConsumer(rt, logger)
	go consumer.Run(ctx) // 内部 ticker 轮询 + 长任务信号量

	// 2) 定时任务（cron）
	c := cron.New(cron.WithSeconds())
	// 每日 02:00 全量对账（AUTH-002 / V1-A4：连续 2 次失败 → P1 告警）
	must(c, "0 0 2 * * *", func() { worker.NewReconciler(rt, logger).RunOnce(ctx) })
	// 每 5 分钟：重试队列扫描（指数退避 1m/5m/15m/1h ≤16 次 → 死信）
	must(c, "0 */5 * * * *", func() { worker.NewRetrofit(rt, logger).Sweep(ctx) })
	// 每 10 分钟：Football 写补偿（ims_fb_sync_outbox）
	must(c, "0 */10 * * * *", func() { worker.NewFootballCompensator(rt, logger).Sweep(ctx) })
	c.Start()

	logger.Info("ims-worker started")
	quit := make(chan os.Signal, 1)
	signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)
	<-quit
	logger.Info("shutdown signal received...")

	c.Stop()
	cancel()
	consumer.Wait() // 等待在途事件处理完成（不丢事件）
	logger.Info("ims-worker stopped")
}

func must(c *cron.Cron, spec string, fn func()) {
	if _, err := c.AddFunc(spec, fn); err != nil {
		panic("cron add failed: " + err.Error())
	}
}
```

### 3.1 `internal/worker/retrofit.go` —— 指数退避 + 死信（骨架）

```go
// internal/worker/retrofit.go
package worker

import (
	"context"
	"time"

	"go.uber.org/zap"
)

const (
	maxRetry         = 16                                   // ≤16 次（V1-A3）
	backoffSchedule  = "1m,5m,15m,1h,1h,1h,1h,1h,1h,1h,1h,1h,1h,1h,1h,1h"
	deadLetterAlert  = "ims.org.dead_letter"
)

// Sweep 扫描到期待重试事件，按退避表推进，超限置死信 + 告警。
func (r *Retrofit) Sweep(ctx context.Context) {
	events, err := r.rt.DBs.ImsQ.ListDueRetryEvents(ctx, r.rt.NowUnix())
	if err != nil {
		r.logger.Error("list due retry events failed", zap.Error(err))
		return
	}
	for _, ev := range events {
		if ev.RetryCount >= maxRetry {
			// 死信：is_dead_letter = true + 告警（V1-A3）
			_ = r.rt.DBs.ImsQ.MarkDeadLetter(ctx, ev.ID)
			r.alert.Fire(ctx, deadLetterAlert, map[string]any{"eventId": ev.DingtalkEventId})
			continue
		}
		next := time.Now().Add(backoffDuration(int(ev.RetryCount)))
		_ = r.rt.DBs.ImsQ.ScheduleRetry(ctx, ev.ID, next) // retry_count++、next_retry_at=next、sync_status=FAILED_RETRY
	}
}

// backoffDuration 返回第 n 次重试的等待时长（1m/5m/15m/1h，n>=3 恒 1h→但 ≤16 次封顶）。
func backoffDuration(n int) time.Duration {
	switch {
	case n <= 0:
		return time.Minute
	case n == 1:
		return 5 * time.Minute
	case n == 2:
		return 15 * time.Minute
	default:
		return time.Hour
	}
}
```

### 4.3 分页 `PageParam` / `PageResult[T]`（`internal/pkg/pagination/page.go`）

> **强制**：`pageNo` 从 1、`pageSize` 默认 10 上限 100；**禁** `page`/`size`/`pageNum`/`limit`/`offset`（全局规范 §2.1/§2.5）。

```go
// internal/pkg/pagination/page.go
package pagination

import (
	"net/http"
	"strconv"
	"strings"
)

const (
	defaultPageNo   = 1
	defaultPageSize = 10
	maxPageSize     = 100
)

// PageParam 分页请求（Query 参数平铺 camelCase）
type PageParam struct {
	PageNo   int `json:"pageNo"`   // 从 1 起，默认 1
	PageSize int `json:"pageSize"` // 默认 10，上限 100
}

// PageResult 分页响应包裹
type PageResult[T any] struct {
	List     []T `json:"list"`     // 当前页数据（空页返回 [] 而非 null）
	Total    int `json:"total"`    // 总条数
	PageNo   int `json:"pageNo"`   // 回显页码
	PageSize int `json:"pageSize"` // 回显每页条数
}

// BindPageParam 从 Query 绑定并做统一校验；禁用变体名（命中即报错）。
func BindPageParam(r *http.Request) (PageParam, error) {
	q := r.URL.Query()
	for _, forbidden := range []string{"page", "size", "pageNum", "limit", "offset"} {
		if q.Has(forbidden) {
			return PageParam{}, &BindError{Field: forbidden}
		}
	}
	p := PageParam{PageNo: defaultPageNo, PageSize: defaultPageSize}
	if v := strings.TrimSpace(q.Get("pageNo")); v != "" {
		n, err := strconv.Atoi(v)
		if err != nil || n < 1 {
			return PageParam{}, &BindError{Field: "pageNo"}
		}
		p.PageNo = n
	}
	if v := strings.TrimSpace(q.Get("pageSize")); v != "" {
		n, err := strconv.Atoi(v)
		if err != nil || n < 1 {
			return PageParam{}, &BindError{Field: "pageSize"}
		}
		if n > maxPageSize {
			n = maxPageSize // 超上限钳制（不报错，与全局规范一致）
		}
		p.PageSize = n
	}
	return p, nil
}

// Offset 供 SQL LIMIT ? OFFSET ? 使用
func (p PageParam) Offset() int { return (p.PageNo - 1) * p.PageSize }

// NewPageResult 构造分页响应（list 判空转为 []）
func NewPageResult[T any](list []T, total int, p PageParam) PageResult[T] {
	if list == nil {
		list = []T{}
	}
	return PageResult[T]{List: list, Total: total, PageNo: p.PageNo, PageSize: p.PageSize}
}

type BindError struct{ Field string }

func (e *BindError) Error() string { return "invalid pagination param: " + e.Field }
```

```go
// internal/authmw/scope_registry.go
package authmw

import "sync"

// ColKind 维度列的匹配方式（D-13）：决定该维度如何生成谓词。
type ColKind int

const (
	ColScalar    ColKind = iota // 普通标量列：col IN (...) / col = ?
	ColJSONArray                // JSON 数组列：JSON_CONTAINS(col, CAST(? AS JSON))（无法走普通索引）
	ColRelation                 // 关联表：EXISTS(rel.fk = t.pk AND rel.value IN (...))（索引友好，推荐）
)

// RelationSpec 关系型维度的关联描述（ColKind==ColRelation 时必填）
type RelationSpec struct {
	Table    string // 关联表名，如 "ims_auth_user_dept"
	FkCol    string // 关联表指向主表的列，如 "user_id"
	ValueCol string // 关联表中承载维度值的列，如 "dept_id"
}

// FilterSpec 单个数据源的过滤列映射（登记册第 10 列落地）
// 说明：列名与匹配方式由注册表统一提供，业务代码不得自带过滤列/不得手写 tenant_id。
// 主表在 SQL 中统一以别名 t 引用（Predicate/EXISTS 依赖）。
type FilterSpec struct {
	Table           string // 逻辑表名（如 "oa_production_content" / "ims_auth_user_mapping"）
	TenantCol       string // 租户列（正交强制轴；通常 "tenant_id"）
	DeptCol         string // 部门列/维度名（SCOPE=DEPT 用；空表示该表不支持 DEPT）
	DeptKind        ColKind // 部门维度匹配方式（默认 ColScalar）
	DeptRel         *RelationSpec // DeptKind==ColRelation 时的关联描述
	IPGroupCol      string // IP 组列（SCOPE=IP_GROUP 用，如 "ip_group_id"）
	IPGroupKind     ColKind // IP 组维度匹配方式（默认 ColScalar）
	CreatorCol      string // 创建者列（SCOPE=SELF 用，如 "creator_id"）
	PublishStateCol string // 发布态列（BR-212 用，空=该表无发布/分享语义）
	ShareCol        string // 分享归属列（BR-212 用，如 "share_owner_id"）
}

// ScopeRegistry 过滤规则注册表（进程内只读；启动期 Register，运行期 Lookup）
type ScopeRegistry struct {
	mu    sync.RWMutex
	specs map[string]FilterSpec
}

func NewScopeRegistry() *ScopeRegistry {
	return &ScopeRegistry{specs: make(map[string]FilterSpec, 64)}
}

// Register 注册（重复表名 panic，强制显式管理）
func (r *ScopeRegistry) Register(spec FilterSpec) {
	r.mu.Lock()
	defer r.mu.Unlock()
	if _, dup := r.specs[spec.Table]; dup {
		panic("duplicate scope registration: " + spec.Table)
	}
	r.specs[spec.Table] = spec
}

// Lookup 查表（未登记 → 返回 false → 中间件 fail-closed 拒绝，B9-6）
func (r *ScopeRegistry) Lookup(table string) (FilterSpec, bool) {
	r.mu.RLock()
	defer r.mu.RUnlock()
	s, ok := r.specs[table]
	return s, ok
}

// registerAll 在启动期调用（示例；后续模块按登记册第 10 列逐表登记）
func registerAll(r *ScopeRegistry) {

	// r.Register(FilterSpec{Table: "ims_xxx", TenantCol: "tenant_id",
	//   DeptCol: "dept_ids", DeptKind: ColJSONArray, CreatorCol: "creator_id"})
```

```go
func maxScopeRank(s ctxutil.DataScope) int {

}

// Resolve 核心：并集求 maxScope（仅 ENABLED 角色参与）；空有效角色 → fail-closed
func (r *Resolver) Resolve(ctx context.Context, lc *ctxutil.LoginContext, spec FilterSpec) (*RowScope, error) {
	if lc.FailClosed() {
		return nil, ctxutil.ErrNoDataPermission // fail-closed（B9-4）
	}
	roles, err := r.roleLoader.LoadEnabledRoles(ctx, lc.TenantID, lc.RoleIDs)
	if err != nil {
		return nil, err
	}
	// 仅 ENABLED 参与；PENDING_CONFIG 空权限角色不得贡献范围（G3）
	var best ctxutil.DataScope
	bestRank := 0
	enabledCount := 0
	for _, rl := range roles {
		if rl.Status != "ENABLED" {
			continue // PENDING_CONFIG 跳过（fail-closed：既不贡献范围也不放行）
		}
		enabledCount++
		if rank := maxScopeRank(rl.DataScope); rank > bestRank {
			bestRank, best = rank, rl.DataScope
		}
	}
	if enabledCount == 0 || bestRank == 0 {
		return nil, ctxutil.ErrNoDataPermission // 无有效角色 / 空权限 → 拒绝
	}

	rs := &RowScope{Scope: best, SelfUserID: lc.UserID, ViewerDeptIDs: lc.DeptIDs, ViewerIPGroupIDs: lc.IPGroupIDs}
	switch best {
	case ctxutil.ScopeAll:
		rs.All = true
	case ctxutil.ScopeDept:
		if spec.DeptCol == "" { // 表不支持 DEPT 轴 → fail-closed
			return nil, ctxutil.ErrNoDataPermission
		}
		rs.DeptIDs = lc.DeptIDs // 不含下级
	case ctxutil.ScopeIPGroup:
		if spec.IPGroupCol == "" {
			return nil, ctxutil.ErrNoDataPermission
		}
		rs.IPGroupIDs = lc.IPGroupIDs // 登录上下文缓存解析（U-4）
	case ctxutil.ScopeSelf:
		if spec.CreatorCol == "" {
			return nil, ctxutil.ErrNoDataPermission
		}
	}
	return rs, nil
}

// Predicate 生成 SQL 谓词：① tenant 正交轴 + ② 查看者维度（四级 dataScope）+ ③ 创建者维度（BR-212）。
// 业务 repo 只需在主表上取别名 t 并把该谓词拼到 WHERE 末尾；绝不自写 tenant_id/dataScope。
// 注意：调用方须将目标表别名固定为 t（EXISTS 子查询以 t.<creator_col> 关联主行）。
func (rs *RowScope) Predicate(tenantID int64, spec FilterSpec) ctxutil.ScopePredicate {
	var sb strings.Builder
	args := make([]any, 0, 12)

	// ① tenant 正交强制轴（B9-1）
	sb.WriteString(fmt.Sprintf(" AND %s = ?", col(spec.TenantCol, "tenant_id")))
	args = append(args, tenantID)

	// ② 查看者维度：行必须落在【查看者】数据范围内（B9-2，四级单轴）
	sb.WriteString(" AND (")
	sb.WriteString(rs.viewerPredicate(spec, &args))
	sb.WriteString(" )")

	// ③ 创建者维度（BR-212）：含"发布态/分享"语义的表，追加——【查看者】必须落在【创建者】数据范围内。
	//    与 ② 为【AND】（两个维度各自满足才可见），**不使用任何顶层 OR 短路**：
	//    短路会让"非发布行"绕过创建者维度而放大可见性（D-12）。
	if spec.PublishStateCol != "" && spec.ShareCol != "" {
		sb.WriteString(" AND ")
		sb.WriteString(rs.creatorDimensionExists(tenantID, spec, &args))
	}

	return ctxutil.ScopePredicate{SQL: sb.String(), Args: args}
}

// viewerPredicate 查看者维度：行落在【查看者】数据范围内的谓词片段（就地 append 参数）。
//   ALL     → 1=1（仅 tenant 轴）
//   DEPT    → 行部门 ∈ 查看者部门集（按 spec.DeptKind 分派：标量列 / JSON 列 / 关联表；不含下级）
//   IP_GROUP→ 行 IP 组 ∈ 查看者 IP 组集
//   SELF    → 行创建者 = 查看者 userId
func (rs *RowScope) viewerPredicate(spec FilterSpec, args *[]any) string {
	switch {
	case rs.All:
		return "1=1"
	case rs.Scope == ctxutil.ScopeDept && spec.DeptCol != "":
		return dimensionPredicate(spec.DeptCol, spec.DeptKind, spec.DeptRel, rs.DeptIDs, args)
	case rs.Scope == ctxutil.ScopeIPGroup && spec.IPGroupCol != "":
		return dimensionPredicate(spec.IPGroupCol, spec.IPGroupKind, nil, rs.IPGroupIDs, args)
	default: // SELF
		*args = append(*args, rs.SelfUserID)
		return fmt.Sprintf("t.%s = ?", spec.CreatorCol)
	}
}

// dimensionPredicate 按 ColKind 生成单个维度的谓词片段（D-13）：
//   ColScalar    → t.<col> IN (?,?,...)
//   ColJSONArray → (JSON_CONTAINS(t.<col>, CAST(? AS JSON)) OR ...)（逐值 OR；无法走普通索引，仅兜底）
//   ColRelation  → EXISTS(SELECT 1 FROM <rel.Table> r WHERE r.<FkCol> = t.id AND r.<ValueCol> IN (?,?,...))
// 空集合一律返回 "1=0"（fail-closed，绝不降级为放行）。
func dimensionPredicate(col string, kind ColKind, rel *RelationSpec, vals []int64, args *[]any) string {
	if len(vals) == 0 {
		return "1=0"
	}
	switch kind {
	case ColJSONArray:
		parts := make([]string, 0, len(vals))
		for _, v := range vals {
			parts = append(parts, fmt.Sprintf("JSON_CONTAINS(t.%s, CAST(? AS JSON))", col))
			*args = append(*args, strconv.FormatInt(v, 10)) // JSON 数字字面量按字符串传参，由 CAST(? AS JSON) 解析
		}
		return "(" + strings.Join(parts, " OR ") + ")"
	case ColRelation:
		r := rel
		if r == nil { // 注册缺失 → fail-closed
			return "1=0"
		}
		*args = append(*args, toAny(vals)...)
		return fmt.Sprintf("EXISTS (SELECT 1 FROM %s r WHERE r.%s = t.id AND r.%s IN (%s))",
			r.Table, r.FkCol, r.ValueCol, placeholders(len(vals)))
	default: // ColScalar
		*args = append(*args, toAny(vals)...)
		return fmt.Sprintf("t.%s IN (%s)", col, placeholders(len(vals)))
	}
}

// creatorDimensionExists 创建者维度（BR-212）：判定【查看者】是否落在【创建者】的数据范围内。
// 通过 ims_user_scope（用户→范围解析视图：由 ims_sys_user_role × 角色 data_scope 物化）做 EXISTS；
// 关联主行的创建者列 t.<creator_col>，与查看者维度 AND，二者各自满足才可见（交集语义）。
//   创建者 scope=ALL     → 任意查看者可见
//   创建者 scope=DEPT    → 查看者任一部门 ∈ 创建者部门集
//   创建者 scope=IP_GROUP→ 查看者任一 IP 组 ∈ 创建者 IP 组集
//   创建者 scope=SELF    → 查看者 == 创建者本人
func (rs *RowScope) creatorDimensionExists(tenantID int64, spec FilterSpec, args *[]any) string {
	var sb strings.Builder
	sb.WriteString("EXISTS (SELECT 1 FROM ims_user_scope cs WHERE ")
	sb.WriteString(fmt.Sprintf("cs.user_id = t.%s AND cs.tenant_id = ?", spec.CreatorCol))
	*args = append(*args, tenantID)
	sb.WriteString(" AND (cs.scope_type = 'ALL'")

	if len(rs.ViewerDeptIDs) > 0 {
		sb.WriteString(fmt.Sprintf(" OR (cs.scope_type = 'DEPT' AND cs.dept_id IN (%s))", placeholders(len(rs.ViewerDeptIDs))))
		*args = append(*args, toAny(rs.ViewerDeptIDs)...)
	}
	if len(rs.ViewerIPGroupIDs) > 0 {
		sb.WriteString(fmt.Sprintf(" OR (cs.scope_type = 'IP_GROUP' AND cs.ip_group_id IN (%s))", placeholders(len(rs.ViewerIPGroupIDs))))
		*args = append(*args, toAny(rs.ViewerIPGroupIDs)...)
	}
	sb.WriteString(" OR (cs.scope_type = 'SELF' AND cs.user_id = ?)")
	*args = append(*args, rs.SelfUserID)
	sb.WriteString(" ) )")
	return sb.String()
}

// Intersect 集合代数参考：RowScope' = 查看者范围 ∩ 创建者范围（B9-3 语义说明用；运行期由
// Predicate 的两维度 AND 表达，无需据此二次求交）。ALL ∩ X = X。
func (rs *RowScope) Intersect(other *RowScope) *RowScope {
	if rs.All {
		return other
	}
	if other.All {
		return rs
	}
	out := &RowScope{SelfUserID: rs.SelfUserID, Scope: rs.Scope}
	out.DeptIDs = intersect64(rs.DeptIDs, other.DeptIDs)
	out.IPGroupIDs = intersect64(rs.IPGroupIDs, other.IPGroupIDs)
	return out
}

// ── 中间件装配 ──

// RowScope 中间件：解析 → 注入 ctx；失败 fail-closed → 403/1008
// 目标表由 handler 通过 X-Scope-Table 或路由元数据声明；此处按路由上下文取。
func RowScope(resolver *Resolver, reg *ScopeRegistry) func(http.Handler) http.Handler {
	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			lc, ok := ctxutil.LoginFrom(r.Context())
			if !ok || lc.FailClosed() {
				response.Fail(w, errcode.ErrNoDataPermission, "no data permission")
				return
			}
			// 目标表来自 handler 注册的 scope 元数据（缺省则只需 tenant 轴）
			table := TableFromContext(r.Context())
			spec := FilterSpec{TenantCol: "tenant_id"}
			if table != "" {
				s, found := reg.Lookup(table)
				if !found {
					// 未登记的数据源 → fail-closed（B9-6）
					response.Fail(w, errcode.ErrNoDataPermission, "data source not registered in scope registry")
					return
				}
				spec = s
			}
			rs, err := resolver.Resolve(r.Context(), lc, spec)
			if err != nil {
				if code, ok := ctxutil.FailCode(err); ok {
					response.Fail(w, code, "no data permission")
					return
				}
				response.Fail(w, errcode.ErrSystemDB, err.Error())
				return
			}
			pred := rs.Predicate(lc.TenantID, spec)
			ctx := ctxutil.WithScopePredicate(r.Context(), &pred)
			next.ServeHTTP(w, r.WithContext(ctx))
		})
	}
}

// ── 工具 ──

func col(v, fallback string) string {
	if v == "" {
		return fallback
	}
	return v
}

func placeholders(n int) string {
	if n <= 0 {
		return "NULL"
	}
	return strings.TrimSuffix(strings.Repeat("?,", n), ",")
}

func toAny(ids []int64) []any {
	out := make([]any, len(ids))
	for i, v := range ids {
		out[i] = v
	}
	return out
}

func intersect64(a, b []int64) []int64 {
	if len(a) == 0 || len(b) == 0 {
		return nil
	}
	set := make(map[int64]struct{}, len(b))
	for _, v := range b {
		set[v] = struct{}{}
	}
	out := make([]int64, 0, len(a))
	for _, v := range a {
		if _, ok := set[v]; ok {
			out = append(out, v)
		}
	}
	return out
}
```

#### 5.3.1 BR-212 交集谓词的可验证示例（硬指标③）

> 用例对齐 QA `ST-B9-030`（U_C 打开 U_A 的 PUBLISHED 分享行 → **被过滤**）。
> 夹具：`T0`；查看者 `U_C`（部门 `D2`，角色 `dataScope=DEPT`，`deptIds=[D2]`）；创建者 `U_A`（部门 `D1`，`dataScope=DEPT`，`deptIds=[D1]`）；目标行 `id=100`（`tenant_id=0, dept_id=D1, creator_id=U_A, publish_state='PUBLISHED', share_owner_id=U_A`）。

**① 中间件为 `oa_production_content`（`FilterSpec` 含 `PublishStateCol/ShareCol`）生成的谓词：**

```sql
SELECT * FROM oa_production_content t
WHERE 1 = 1
  AND t.tenant_id = ?                                            -- ① tenant 正交轴
  AND ( t.dept_id IN (?) )                                       -- ② 查看者维度（U_C=DEPT:[D2]）
  AND EXISTS (                                                   -- ③ 创建者维度（BR-212）
        SELECT 1 FROM ims_user_scope cs
        WHERE cs.user_id = t.creator_id AND cs.tenant_id = ?
          AND ( cs.scope_type = 'ALL'
             OR (cs.scope_type = 'DEPT'     AND cs.dept_id     IN (?))   -- 查看者部门 [D2]
             OR (cs.scope_type = 'SELF'     AND cs.user_id      = ?)   )-- 查看者 U_C
      )
ORDER BY t.id DESC LIMIT ? OFFSET ?;
-- 绑定参数（与谓词出现顺序一致）：[T0, D2, T0, D2, U_C]
```

**② 预期结果：**

| 检查 | 结果 | 判定 |
|------|------|------|
| 明文「查看者维度」 | `t.dept_id=D1` ∈ `{D2}` → **false** | 已过滤 |
| 「创建者维度」EXISTS | 创建者 U_A 的 `scope=DEPT;dept_id=D1`；查看者部门 `{D2}` 不含 `D1` → **无匹配行** → false | 已过滤 |
| 最终 | 两维度 **AND → false** | **返回 0 行：U_C 看不到 U_A 的分享行** ✅ |

**③ 对照正确性（反例）**：若查看者为 `U_A` 本人（`dept_id=D1`），则 ② 命中 `D1∈{D1}`、③ EXISTS 命中 `scope=DEPT & dept_id=D1∈{D1}` → **两维度均 true → 可见**。若沿用旧谓词 `(publish_state<>'PUBLISHED' OR share_owner=self)`，`U_C` 命中 `share_owner_id=U_A ≠ U_C` 且 `publish_state='PUBLISHED'` → 亦过滤——**但旧谓词未表达的「两集合求交」在跨部门、`ALL`×`DEPT` 等组合下会错误放行/漏放**（如 `ALL` 查看者本应看到交集、旧谓词却把已发布行一股脑限为「查看者=本人」）。

**④ 与四级 dataScope 的关系（两维度分别怎么算）：**

| 维度 | 语义 | 计算来源 | 四级落点 |
|------|------|----------|----------|
| **查看者维度**（②） | 目标**行**落在**查看者**可见范围内 | 查看者角色的 `dataScope`（`RowScope`） | `ALL`→不限制；`DEPT`→`row.dept ∈ viewer.deptIds`（不含下级）；`IP_GROUP`→`row.ip_group ∈ viewer.ipGroups`；`SELF`→`row.creator = viewer.userId` |
| **创建者维度**（③） | **查看者**落在**行的创建者**可见范围内 | 创建者角色的 `dataScope`（`ims_user_scope` 物化） | `ALL`→任意查看者；`DEPT`→`viewer.dept ∩ creator.dept ≠ ∅`；`IP_GROUP`→`viewer.ipGroup ∩ creator.ipGroup ≠ ∅`；`SELF`→`viewer == creator` |

> 二者**各自独立求值后 AND**（交集语义）：任一方为假即不可见，**无 OR 短路**。草稿（`DRAFT`）行同受两维度约束——方向恒为 **fail-closed**（只会更严格，绝不因分享放大可见性）。

#### 5.3.2 `dept_ids` 为 JSON 列时的 DEPT 过滤方案（D-13）

> **问题**：`ims_auth_user_mapping.dept_ids` 是 **JSON 数组列**，旧骨架直接生成 `dept_ids IN (?,?)` **语义错误**（JSON 列不能直接 `IN`）。

**方案对比与选型**：

| 方案 | 谓词 | 索引 | 结论 |
|------|------|------|------|
| ① `JSON_CONTAINS` | `JSON_CONTAINS(t.dept_ids, CAST(? AS JSON))`（逐值 OR） | 依赖 MySQL 8.0.17+ **多值索引** `CAST(dept_ids AS UNSIGNED ARRAY)`；否则**全表扫描** | 仅作**兜底**（无法加关联表的临时表） |
| ② **关联表（推荐）** | `EXISTS(SELECT 1 FROM ims_auth_user_dept r WHERE r.user_id = t.id AND r.dept_id IN (?,?))` | 关联表建 `(dept_id, user_id)` 复合索引，**可走索引**、可统计、可对账 | **采用** |

**落地（已写入骨架）**：
- `FilterSpec` 增 `DeptKind`（`ColScalar`/`ColJSONArray`/`ColRelation`）+ `DeptRel *RelationSpec`；`registerAll` 中 `ims_auth_user_mapping` 声明 `DeptKind=ColRelation, DeptRel={ims_auth_user_dept, user_id, dept_id}`。
- `dimensionPredicate()` 按 `ColKind` 分派：标量 `IN`、JSON `JSON_CONTAINS`、关联 `EXISTS`——**与 `scope_registry` 的列映射机制一致**（登记即生效，业务代码零改动）。
- **关联表物化**：`ims_auth_user_dept(user_id, dept_id)` 由 `dept_ids` 在**同一事务**写时同步（upsert 时先删后插），保证与 JSON 列一致；建 `PRIMARY KEY(user_id, dept_id)` + `KEY idx_dept_user(dept_id, user_id)`。
- **索引策略**：DEPT 过滤走 `idx_dept_user`（`dept_id` 前缀），`EXISTS` 半连接可下推；`JSON_CONTAINS` 兜底方案须显式建多值索引，否则登记表须限制数据量。

**DEPT「不含下级」验证说明（对齐 ST-B9-020 / 无 `DEPT_AND_CHILD`）**：
- 谓词恒为 `dept_id ∈ ctx.deptIds`（**精确命中**），**不存在**任何「取子部门闭包 / 递归 CTE / `parent_id` 展开」的分支——`DEPT_AND_CHILD` 已废弃，代码中**无该枚举值**。
- 验证：夹具 `D1`（父）、`D1-1`（D1 子）。`DEPT` 用户 `U_A(D1)` 查询结果**不含** `dept_id=D1-1` 的行；`U_A(D1)` 与 `U_B(D1-1)` 结果集的部门集合交集 = ∅（`ST-B9-020` 反证通过）。
- 关联表方案下同样精确：`r.dept_id IN (D1)` 不会因父子关系命中 `D1-1`。

### 5.5 `guard.go` / `scopedrepo` —— 跨租户统一 1504（B9-5 · **一处执行**，D-17）

> **D-17 问题**：旧骨架把 1504 交由 per-handler `EnsureTenant` 触发，中间件链自身不产出 1504 → 与 B9-5「一处执行」矛盾、逐 handler 手写易漏。
> **修正**：提供**两个统一拦截点**（handler 不手写）——
> **(A) DAO 层公共包装 `scopedrepo.MustOwn`**（按 id 取/改单条资源的**主用**通道）；
> **(B) chi 路由中间件 `TenantResourceGuard`**（对 `/.../{id}` 类路由，在进入 handler 前预取 `tenant_id` 拦截）。

```go
// internal/authmw/guard.go

// EnsureTenant 校验目标资源 tenant_id 与上下文一致；不一致统一返回 1504（不泄露存在性）。
func EnsureTenant(ctx context.Context, resourceTenantID int64) error {
	lc, err := ctxutil.MustLogin(ctx)
	if err != nil {
		return ctxutil.ErrNoDataPermission
	}
	if resourceTenantID != lc.TenantID {
		return ctxutil.ErrCrossTenant // → 1504
	}
	return nil
}

// WriteErr 统一错误映射（handler 只调此一处，不各自判码）：
//   1008 无数据权限 / 1504 跨租户 / 1500 不存在 / 5001 兜底。跨租户与不存在的 msg 均为通用文案，不含 id/名称。
func WriteErr(w http.ResponseWriter, err error) {
	switch {
	case errors.Is(err, ctxutil.ErrNoDataPermission):
		response.Fail(w, errcode.ErrNoDataPermission, "no data permission")
	case errors.Is(err, ctxutil.ErrCrossTenant):
		response.Fail(w, errcode.ErrCrossTenant, "resource not accessible") // 通用文案，无存在性信息
	case errors.Is(err, ctxutil.ErrNotFound), errors.Is(err, sql.ErrNoRows):
		response.Fail(w, errcode.ErrNotFound, "resource not found")
	default:
		response.Fail(w, errcode.ErrSystemDB, "internal error")
	}
}

// ── (B) 路由 → 资源表注册 + 中间件 ──

// resourceRoutes 路由模式（含 {id}）→ 资源表 映射（启动期登记）。
type resourceRoutes struct {
	mu     sync.RWMutex
	byPath map[string]string
}

func newResourceRoutes() *resourceRoutes { return &resourceRoutes{byPath: make(map[string]string, 32)} }

func (rr *resourceRoutes) register(pattern, table string) {
	rr.mu.Lock()
	defer rr.mu.Unlock()
	rr.byPath[pattern] = table
}

func (rr *resourceRoutes) lookup(pattern string) (string, bool) {
	rr.mu.RLock()
	defer rr.mu.RUnlock()
	t, ok := rr.byPath[pattern]
	return t, ok
}

// TenantResourceGuard 对 {id} 路由在 handler 前预取 tenant_id；跨租户 → 1504（不进业务逻辑）。
// D-24（fail-closed）：**未登记的 {id} 路由一律拒绝**（不再静默放行）；无 {id} 的路由才放行。
func TenantResourceGuard(rr *resourceRoutes, db *sql.DB, logger *zap.Logger) func(http.Handler) http.Handler {
	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			pattern := chi.RouteContext(r.Context()).RoutePattern() // 如 /admin-api/ims/system/role/{id}/data-scope
			table, ok := rr.lookup(pattern)
			if !ok {
				if strings.Contains(pattern, "{id}") {
					// 含 {id} 却未登记 → fail-closed 拒绝 + 告警（漏登记即静默失守的根因被堵死）
					logger.Error("resource route not registered (fail-closed)",
						zap.String("pattern", pattern))
					response.Fail(w, errcode.ErrSystemDB, "resource route not registered")
					return
				}
				next.ServeHTTP(w, r) // 无 {id} → 无租户资源，放行
				return
			}
			id, err := strconv.ParseInt(chi.URLParam(r, "id"), 10, 64)
			if err != nil {
				response.Fail(w, errcode.ErrAuthParamInvalid, "invalid id")
				return
			}
			if err := scopedrepo.MustOwn(r.Context(), db, table, id); err != nil {
				WriteErr(w, err) // 1504 / 1500，响应不含目标资源存在性
				return
			}
			next.ServeHTTP(w, r)
		})
	}
}

// ValidateAllIDRoutes 启动期校验：遍历全部路由，凡含 {id} 的必须已登记（否则 Fatal，避免开发期大面积阻塞）。
// 与 TenantResourceGuard 的 fail-closed 形成「启动拦截 + 运行拦截」双保险。
func ValidateAllIDRoutes(router chi.Routes, rr *resourceRoutes) error {
	return chi.Walk(router, func(method, route string, _ http.Handler, _ ...func(http.Handler) http.Handler) error {
		if strings.Contains(route, "{id}") {
			if _, ok := rr.lookup(route); !ok {
				return fmt.Errorf("route %s %s 含 {id} 但未在 resourceRoutes 登记（fail-closed）", method, route)
			}
		}
		return nil
	})
}
```

### 6.1 `internal/domain/auth/sso.go`（服务骨架）

```go
// internal/domain/auth/sso.go
package auth

import (
	"context"
	"crypto/rand"
	"encoding/hex"

	"time"

	"github.com/redis/go-redis/v9"

	"github.com/shenyu/ims/internal/pkg/dingtalk"

const (
	stateTTL        = 5 * time.Minute   // state 防 CSRF（AUTH-001）
	accessTokenTTL  = 2 * time.Hour     // AUTH-R2
	refreshTokenTTL = 7 * 24 * time.Hour
	maxConcurrent   = 3                 // AUTH-R3：并发会话 ≤ 3（对齐 AUTH 契约；实现见 enforceConcurrent/D-15）
	lockFailLimit   = 5                 // AUTH-R4：连续失败 5 次（实现见 checkLoginLock/incrLoginFail/D-14）
	lockDuration    = 15 * time.Minute  // AUTH-R4：锁定时长
)

type SSOService struct {
	rdb      *redis.Client
	dt       *dingtalk.Client
	fb       *football.FootballWebApiClient // 唯一外部 HTTP 白名单（Football）
	signer   *JWTSigner
	mappings UserMappingRepo
}

// Redirect 生成钉钉授权 URL + state（存 Redis 5min）
func (s *SSOService) Redirect(ctx context.Context, redirectURI string) (SsoRedirectResp, error) {
	state, err := randHex(16)
	if err != nil {
		return SsoRedirectResp{}, err
	}
	if err := s.rdb.Set(ctx, "sso:state:"+state, "1", stateTTL).Err(); err != nil {
		return SsoRedirectResp{}, errcode.Wrap(errcode.ErrRedis, err)
	}
	url := s.dt.BuildAuthorizeURL(state, redirectURI)
	return SsoRedirectResp{AuthorizeURL: url, State: state}, nil
}

// Callback 授权码换 JWT（state 校验 → 钉钉换用户 → 登录锁定校验 → BR-308 mobile 命中 → 并发会话 → 签发）
// 本函数实现 AUTH-R4「连续失败 5 次锁 15min」（D-14）。
func (s *SSOService) Callback(ctx context.Context, req SsoCallbackReq) (SsoCallbackResp, error) {
	// ① state 一次性校验（DEL 保证不可重放）
	ok, err := s.rdb.Del(ctx, "sso:state:"+req.State).Result()
	if err != nil || ok == 0 {
		return SsoCallbackResp{}, errcode.New(errcode.ErrSessionInvalid, "state 无效或已过期")
	}

	// ② 钉钉授权码换用户（超时/限流 → 5003）
	dingUser, err := s.dt.GetUserByCode(ctx, req.AuthCode)
	if err != nil {
		return SsoCallbackResp{}, errcode.New(errcode.ErrDingtalkTimeout, "钉钉暂不可用")
	}

	// ②' 登录主体（AUTH-R4 计数键；优先 mobile，缺省退 unionId）
	subject := loginSubject(dingUser)

	// ③ 锁定校验（AUTH-R4）：连续失败达 5 次 → 锁定期间一律拦截，返回 1001 附 R4 说明
	if locked, remain := s.checkLoginLock(ctx, subject); locked {
		return SsoCallbackResp{}, errcode.New(errcode.ErrAuthParamInvalid,
			fmt.Sprintf("登录失败次数过多，账号已锁定，请 %d 分钟后再试", int(remain.Minutes())+1))
	}

	// ④ BR-308：手机号命中 Football system_users.id（HTTP 读；IMS 不回写用户主表）
	//    无手机号 / 匹配失败 → 触发单用户即时同步（V1-A5）；仍失败 → 1006（不签发 Token）
	if dingUser.Mobile == "" {
		s.incrLoginFail(ctx, subject) // 计入失败（AUTH-R4）
		return SsoCallbackResp{}, errcode.New(errcode.ErrDingtalkSyncFail, "缺少手机号，无法匹配运营用户")
	}
	sysUser, err := s.fb.ListSystemUsersByMobile(ctx, dingUser.Mobile)
	if err != nil || sysUser == nil {
		if syncErr := s.triggerSingleUserSync(ctx, dingUser); syncErr != nil {
			s.incrLoginFail(ctx, subject)
			return SsoCallbackResp{}, errcode.New(errcode.ErrDingtalkSyncFail, "钉钉同步失败")
		}
		sysUser, err = s.fb.ListSystemUsersByMobile(ctx, dingUser.Mobile)
		if err != nil || sysUser == nil {
			s.incrLoginFail(ctx, subject)
			return SsoCallbackResp{}, errcode.New(errcode.ErrDingtalkSyncFail, "钉钉同步失败")
		}
	}

	// ⑤ 离职冻结用户不签发 Token（ORG-R4）
	if sysUser.StatusFrozen() {
		s.incrLoginFail(ctx, subject)
		return SsoCallbackResp{}, errcode.New(errcode.ErrSessionInvalid, "账号已冻结")
	}

	// ⑥ 构建 LoginContext（含角色并集、deptIds、ipGroupIds 缓存解析）
	lc, err := s.buildLoginContext(ctx, sysUser)
	if err != nil {
		s.incrLoginFail(ctx, subject)
		return SsoCallbackResp{}, err
	}

	// ⑦ 并发会话 ≤ 3：Lua 原子「ZAdd 新会话 → 超额踢最早 → 入黑名单」（AUTH-R3；D-15）
	if err := s.enforceConcurrent(ctx, sysUser.ID, lc.SessionID); err != nil {
		return SsoCallbackResp{}, err
	}

	// ⑧ 签发 access(2h) / refresh(7d)
	access, refresh, expiresAt, err := s.signer.Issue(lc)
	if err != nil {
		return SsoCallbackResp{}, err
	}
	// 会话登记 + 登录成功清零失败计数（AUTH-R4）
	_ = s.rdb.Set(ctx, "sso:session:"+lc.SessionID, sysUser.ID, accessTokenTTL).Err()
	s.resetLoginFail(ctx, subject)

	return SsoCallbackResp{
		AccessToken: access, RefreshToken: refresh, ExpiresAt: expiresAt,
		UserProfile: toProfile(sysUser, lc),
	}, nil
}

// Refresh 刷新 Token（校验 refresh → 重新签发 access）
func (s *SSOService) Refresh(ctx context.Context, refreshToken string) (SsoCallbackResp, error) {
	lc, err := s.signer.ParseRefresh(refreshToken)
	if err != nil {
		return SsoCallbackResp{}, errcode.New(errcode.ErrSessionInvalid, "refresh 无效")
	}
	if blacklisted, _ := s.rdb.SIsMember(ctx, "sso:blacklist", lc.RefreshJTI()).Result(); blacklisted {
		return SsoCallbackResp{}, errcode.New(errcode.ErrSessionInvalid, "会话已注销")
	}
	access, refresh, expiresAt, err := s.signer.Issue(lc)
	if err != nil {
		return SsoCallbackResp{}, err
	}
	return SsoCallbackResp{AccessToken: access, RefreshToken: refresh, ExpiresAt: expiresAt, UserProfile: toProfile(nil, lc)}, nil
}

// Logout 注销会话（access jti + refresh jti 入黑名单；移除并发 ZSET 成员）
func (s *SSOService) Logout(ctx context.Context, lc *ctxutil.LoginContext) error {
	_ = s.rdb.SAdd(ctx, "sso:blacklist", lc.AccessJTI(), lc.RefreshJTI()).Err()
	_ = s.rdb.Expire(ctx, "sso:blacklist", refreshTokenTTL).Err()
	_ = s.rdb.Del(ctx, "sso:session:"+lc.SessionID).Err()
	_ = s.rdb.ZRem(ctx, "sso:concurrent:"+itoa(lc.UserID), lc.SessionID).Err()
	return nil
}

// concurrentLua 原子执行 AUTH-R3「并发会话 ≤ 3，超限踢最早」（D-15 修复：此前缺 ZAdd → ZSET 恒空、限流失效）。
// KEYS[1]=sso:concurrent:{userId}  KEYS[2]=sso:blacklist  KEYS[3]=sso:concurrent:seq
// ARGV[1]=新 sessionId/jti  ARGV[2]=登录时间戳(ms)  ARGV[3]=上限(maxConcurrent)  ARGV[4]=TTL(秒)
// 返回：被踢出的 sessionId 列表。
//
// D-25（同毫秒 tie 确定性）：score = 时间戳ms 高位 + 【全局自增序列】低位，保证严格单调递增，
//   同毫秒多次登录也有确定顺序（后来者 score 更大）→「踢最早」语义确定，不会随机踢错。
//   （纯 nowMs 作 score 时，同毫秒 ZRANGE 0 0 取哪条不确定；此处用 Redis INCR 序列消除歧义。）
var concurrentLua = redis.NewScript(`
local seq = redis.call('INCR', KEYS[3])                     -- 全局单调序列（tie-break 低位）
local score = tonumber(ARGV[2]) * 1000 + (seq % 1000)       -- 高位=毫秒时间戳，低位=序列（同毫秒内递增）
redis.call('ZADD', KEYS[1], score, ARGV[1])                 -- ① 新会话入集合（score 严格单调，member=sessionId）
redis.call('EXPIRE', KEYS[3], ARGV[4])                      -- 序列 key 同生命周期
local limit = tonumber(ARGV[3])
local n = redis.call('ZCARD', KEYS[1])
local evicted = {}
while n > limit do                                          -- ② 超出上限 → 循环踢最早，直到 n <= limit
  local oldest = redis.call('ZRANGE', KEYS[1], 0, 0)        --    score 最小 = 最早登录（序列兜底保证唯一）
  if #oldest == 0 then break end
  redis.call('ZREM', KEYS[1], oldest[1])
  redis.call('SADD', KEYS[2], oldest[1])                    -- ③ 被踢会话入黑名单（其 access 再调用 → 1002）
  redis.call('EXPIRE', KEYS[2], ARGV[4])
  table.insert(evicted, oldest[1])
  n = redis.call('ZCARD', KEYS[1])
end
redis.call('EXPIRE', KEYS[1], ARGV[4])                      -- ④ 集合随会话生命周期续期
return evicted
`)

// enforceConcurrent 新登录时调用：保证「登录后活动会话 ≤ maxConcurrent(=3=AUTH-R3 口径)」，超额踢最早。
//
// 原子性：INCR → ZAdd → ZCard → ZRange → ZRem → SAdd 全部在【单个 Lua 脚本】内由 Redis 单线程原子执行，
// 期间不会被其他客户端的并发登录穿插，因此并发下不会出现「计数偏大/偏小」「重复踢同一会话」「漏踢」。
// 不误踢原理：每轮只删 score 最小（最早）成员；score 严格单调 → 本次刚 ZAdd 的新会话 score 最大，永不成为 victim。
func (s *SSOService) enforceConcurrent(ctx context.Context, userID int64, sessionID string) error {
	key := "sso:concurrent:" + itoa(userID)
	nowMs := time.Now().UnixMilli()
	_, err := concurrentLua.Run(ctx, s.rdb,
		[]string{key, "sso:blacklist", "sso:concurrent:seq"},
		sessionID, nowMs, maxConcurrent, int(refreshTokenTTL.Seconds()),
	).Result()
	if err != nil && err != redis.Nil {
		return errcode.Wrap(errcode.ErrRedis, err)
	}
	return nil
}

// ── AUTH-R4 登录失败锁定（D-14）：连续失败 5 次 → 锁 15min；成功清零；TTL 到期自动解锁 ──

const loginFailKeyPrefix = "sso:loginfail:"

// loginSubject 登录主体（锁定计数键，AUTH-R4）；优先手机号，缺省退 unionId。
func loginSubject(u *dingtalk.DingUser) string {
	if u.Mobile != "" {
		return "m:" + u.Mobile
	}
	return "u:" + u.UnionID
}

// checkLoginLock 判定是否处于锁定（连续失败 ≥ lockFailLimit=5）；返回剩余锁定剩余时长。
func (s *SSOService) checkLoginLock(ctx context.Context, subject string) (bool, time.Duration) {
	key := loginFailKeyPrefix + subject
	n, err := s.rdb.Get(ctx, key).Int()
	if err == redis.Nil || n < lockFailLimit {
		return false, 0 // 键不存在 / 未达阈值 → 未锁定
	}
	ttl, _ := s.rdb.TTL(ctx, key).Result()
	if ttl <= 0 {
		return false, 0 // TTL 已过 → 自动解锁
	}
	return true, ttl
}

// incrLoginFail 失败计数：INCR + 每次续期 lockDuration(15min)（滑动窗口 → 锁定后 15min 内无新失败即自动解锁）。
func (s *SSOService) incrLoginFail(ctx context.Context, subject string) {
	key := loginFailKeyPrefix + subject
	pipe := s.rdb.TxPipeline()
	pipe.Incr(ctx, key)
	pipe.Expire(ctx, key, lockDuration) // 首次即设 TTL；后续每次失败续期
	_, _ = pipe.Exec(ctx)
}

// resetLoginFail 登录成功清零失败计数（AUTH-R4 清零时机）。
func (s *SSOService) resetLoginFail(ctx context.Context, subject string) {
	_ = s.rdb.Del(ctx, loginFailKeyPrefix+subject).Err()
}

// buildLoginContext 组装登录上下文（角色并集在 §5 Resolver 侧用于 dataScope）
func (s *SSOService) buildLoginContext(ctx context.Context, u *football.SystemUser) (*ctxutil.LoginContext, error) {
	roleIDs, roleKeys, deptIDs, ipGroupIDs, err := s.mappings.LoadUserRolesAndGroups(ctx, u.ID)
	if err != nil {
		return nil, err
	}
	return &ctxutil.LoginContext{
		UserID: u.ID, TenantID: 0, DeptIDs: deptIDs,
		RoleIDs: roleIDs, RoleKeys: roleKeys,
		IPGroupIDs: ipGroupIDs, Authenticated: true,
		SessionID: randSessionID(),
	}, nil
}

func randHex(n int) (string, error) {
	b := make([]byte, n)
	if _, err := rand.Read(b); err != nil {
		return "", err
	}
	return hex.EncodeToString(b), nil
}
```

### 6.2 D1 钉钉 SSO 回调（`GET /sso/dingtalk/callback`）

```go

		q := r.URL.Query()
		authCode, state := q.Get("authCode"), q.Get("state")
		if authCode == "" || state == "" {
			response.Fail(w, errcode.ErrSessionInvalid, "missing authCode/state")
			return
		}
		// 复用同一 Service；登录渠道 H5（免登）
		_, err := sso.Callback(r.Context(), SsoCallbackReq{
			AuthCode: authCode, State: state, LoginChannel: "H5",
		})
		if err != nil {
			code := errcode.ErrDingtalkSyncFail
			if e, ok := errcode.As(err); ok {
				code = e.Code
			}
			response.Fail(w, code, err.Error())
			return
		}
		http.Redirect(w, r, "/ims/workbench", http.StatusFound)
	}
}
```

### 6.3 JWT 签发（`internal/domain/auth/jwt.go`）

```go
// internal/domain/auth/jwt.go
package auth

import (
	"time"

	"github.com/golang-jwt/jwt/v5"
	"github.com/shenyu/ims/internal/ctxutil"
)

type JWTSigner struct {
	secret      []byte
	issuer      string
	accessTTL   time.Duration // 2h
	refreshTTL  time.Duration // 7d
}

type claims struct {
	UserID    int64    `json:"uid"`
	TenantID  int64    `json:"tid"`
	DeptIDs   []int64  `json:"dids"`
	RoleIDs   []int64  `json:"rids"`
	RoleKeys  []string `json:"rks"`
	IPGroupIDs []int64 `json:"igids"`
	SessionID string   `json:"sid"`
	jwt.RegisteredClaims
}

func (s *JWTSigner) Issue(lc *ctxutil.LoginContext) (access, refresh, expiresAt string, err error) {
	now := time.Now()
	base := jwt.RegisteredClaims{
		Issuer:    s.issuer,
		IssuedAt:  jwt.NewNumericDate(now),
		ID:        lc.SessionID,
	}
	ac := claims{UserID: lc.UserID, TenantID: lc.TenantID, DeptIDs: lc.DeptIDs, RoleIDs: lc.RoleIDs, RoleKeys: lc.RoleKeys, IPGroupIDs: lc.IPGroupIDs, SessionID: lc.SessionID, RegisteredClaims: base}
	ac.ExpiresAt = jwt.NewNumericDate(now.Add(s.accessTTL))
	access, err = jwt.NewWithClaims(jwt.SigningMethodHS256, ac).SignedString(s.secret)
	if err != nil {
		return "", "", "", err
	}
	rc := claims{UserID: lc.UserID, TenantID: lc.TenantID, SessionID: lc.SessionID, RegisteredClaims: jwt.RegisteredClaims{
		Issuer: s.issuer, IssuedAt: jwt.NewNumericDate(now), ID: lc.SessionID + ":r", ExpiresAt: jwt.NewNumericDate(now.Add(s.refreshTTL)),
	}}
	refresh, err = jwt.NewWithClaims(jwt.SigningMethodHS256, rc).SignedString(s.secret)
	return access, refresh, now.Add(s.accessTTL).Format(time.RFC3339), err
}
```

```go
import (
	"context"
	"net/http"
	"strings"

	"github.com/redis/go-redis/v9"
	"github.com/shenyu/ims/internal/config"
	"github.com/shenyu/ims/internal/ctxutil"
	"github.com/shenyu/ims/internal/errcode"
	"github.com/shenyu/ims/internal/pkg/response"
)

// JWTAuth 解析 Bearer → 构建 LoginContext → 注入 ctx；失效 401（前端跳 SSO）
func JWTAuth(cfg config.JWTConfig, rdb *redis.Client) func(http.Handler) http.Handler {
	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			raw := bearer(r)
			if raw == "" {
				response.Fail(w, errcode.ErrSessionInvalid, "missing token")
				return
			}
			cl, err := parseJWT(raw, cfg.Secret)
			if err != nil {
				response.Fail(w, errcode.ErrSessionInvalid, "invalid token")
				return
			}
			// 黑名单校验
			if bl, _ := rdb.SIsMember(r.Context(), "sso:blacklist", cl.SessionID).Result(); bl {
				response.Fail(w, errcode.ErrSessionInvalid, "session revoked")
				return
			}
			lc := &ctxutil.LoginContext{
				UserID: cl.UserID, TenantID: cl.TenantID,
				DeptIDs: cl.DeptIDs, RoleIDs: cl.RoleIDs, RoleKeys: cl.RoleKeys,
				IPGroupIDs: cl.IPGroupIDs, SessionID: cl.SessionID, Authenticated: true,
			}
			next.ServeHTTP(w, r.WithContext(ctxutil.WithLoginContext(r.Context(), lc)))
		})
	}
}

func bearer(r *http.Request) string {
	h := r.Header.Get("Authorization")
	if strings.HasPrefix(h, "Bearer ") {
		return strings.TrimSpace(strings.TrimPrefix(h, "Bearer "))
	}
	return ""
}
```

```ts
  baseURL: '/admin-api/ims',
  timeout: 15000,
})

// 请求拦截：注入 Bearer
http.interceptors.request.use((config) => {
  const token = localStorage.getItem(TOKEN_KEY)
  if (token) {
    config.headers = config.headers ?? ({} as AxiosRequestHeaders)
    config.headers.Authorization = `Bearer ${token}`
  }
  return config
})

// 响应拦截：CommonResult 解包 + 统一错误
http.interceptors.response.use(
  (resp) => {
    const body = resp.data as CommonResult<unknown>
    if (body.code === 0) return body.data as never // 调用方直接拿 data
    // 业务错误：抛出并提示 msg
    ElMessage.error(body.msg || '请求失败')
    return Promise.reject(new BizError(body.code, body.msg))
  },
  (error) => {
    const status = error?.response?.status
    if (status === 401) {
      // 未登录/会话失效 → 跳钉钉 SSO（静默）
      localStorage.removeItem(TOKEN_KEY)
      window.location.href = SSO_REDIRECT_PATH
      return Promise.reject(error)
    }
    if (status === 403) ElMessage.error('无数据权限')
    else ElMessage.error(error?.message || '网络异常')
    return Promise.reject(error)
  },
)

export class BizError extends Error {
```

```ts
// src/stores/user.ts
// 定位：前端权限判断【仅用于 UI 控制（菜单/按钮显隐/路由准入）】，**不承担安全职责**——

import { defineStore } from 'pinia'
import http from '@/utils/request'
import type { UserProfileVO, SsoCallbackResp } from '@/api/types/auth'

const TOKEN_KEY = 'ims_access_token'
const REFRESH_KEY = 'ims_refresh_token'

export const useUserStore = defineStore('user', {
  state: () => ({
    token: localStorage.getItem(TOKEN_KEY) ?? '',
    refreshToken: localStorage.getItem(REFRESH_KEY) ?? '',
    profile: null as UserProfileVO | null,
  }),
  getters: {
    roles: (s) => s.profile?.roleCodes ?? [],
    // 离职冻结（FROZEN，账号态）≠ 待配置（PENDING_CONFIG，角色态）：二者语义不同，禁止等同（D-19 顺带修正）
    isFrozen: (s) => s.profile?.status === 'FROZEN',
  },
  actions: {
    setToken(t: string) { this.token = t; localStorage.setItem(TOKEN_KEY, t) },
    setRefreshToken(t: string) { this.refreshToken = t; localStorage.setItem(REFRESH_KEY, t) },
    clear() {
      this.token = ''
      this.refreshToken = ''
      this.profile = null
      localStorage.removeItem(TOKEN_KEY)
      localStorage.removeItem(REFRESH_KEY)
    },
    async ensureProfile() {
      if (this.profile) return
      // profile 由 SSO callback / refresh 响应携带（契约 SsoCallbackResp.userProfile）。
      // 契约：refresh 为 POST，Body { refreshToken }，响应含新 Token 对 + userProfile（非 GET、非仅 Token）。
      const resp = await http.post('/auth/sso/refresh', {
        refreshToken: this.refreshToken,
      }) as unknown as SsoCallbackResp
      this.setToken(resp.accessToken)
      this.setRefreshToken(resp.refreshToken)
      this.profile = resp.userProfile
    },
    hasPerm(code: string): boolean {
      // fail-closed：仅当权限码集合【明确包含】该码才放行；无明确权限码 → 不渲染 / 不放行
      return usePermissionStore().codes.includes(code)
    },
  },
})

// src/stores/permission.ts
import { defineStore } from 'pinia'
import http from '@/utils/request'
import type { MenuNode } from '@/api/types/sys'

export const usePermissionStore = defineStore('permission', {
  state: () => ({ menus: [] as MenuNode[], codes: [] as string[], loaded: false }),
  actions: {
    async load() {
      if (this.loaded) return
      this.menus = await http.get('/system/menu/tree') as unknown as MenuNode[]
      this.codes = flattenPerms(this.menus)
      this.loaded = true
    },
  },
})

function flattenPerms(nodes: MenuNode[]): string[] {
  const out: string[] = []
  const walk = (ns: MenuNode[]) => ns.forEach((n) => { if (n.permCode) out.push(n.permCode); if (n.children) walk(n.children) })
  walk(nodes)
  return out
}

import { usePermissionStore } from '@/stores/permission'
```

```bash
DENY_HOST='\bops\.(shenyu|internal|local|corp|prod|test)\b|\bops-(gateway|service|api|host)\b|\bOPS_(HOST|GATEWAY|BASE_URL|SERVICE)\b'
# 注意：OPS host 在【任何位置】均禁止（含白名单目录），故不做目录排除
if rg -n --no-heading "$DENY_HOST" "${BACKEND_DIRS[@]}" "${FRONTEND_DIRS[@]}" "${CONF_GLOBS[@]}" deploy 2>/dev/null; then
  echo "✗ 命中 OPS host 引用"; FAIL=1
fi

echo "== [2/5] 禁项二：ops. base-url / OPS RPC 依赖（同样全域禁止）=="
DENY_BASEURL='\bops[._-]base[._-]url\b|\bopsBaseUrl\b|FeignClient|@DS\(|/rpc-api/ops|openfeign'
if rg -n --no-heading "$DENY_BASEURL" "${BACKEND_DIRS[@]}" "${FRONTEND_DIRS[@]}" "${CONF_GLOBS[@]}" deploy 2>/dev/null; then
  echo "✗ 命中 OPS base-url / RPC 依赖（Feign/@DS/openfeign 等）"; FAIL=1
fi

echo "== [3/5] 禁项三：外部 HTTP 调用仅限白名单适配层目录 =="
# 任何外部 HTTP 客户端调用必须落在白名单目录内；其余路径出现即违规
DENY_HTTPCALL='http\.NewRequest\(|http\.Get\(|http\.Post\(|resty\.|req\.NewRequest\('
HITS="$(rg -l "$DENY_HTTPCALL" "${BACKEND_DIRS[@]}" 2>/dev/null || true)"
if [ -n "$HITS" ]; then
  VIOL="$(echo "$HITS" | grep -vE "$WHITELIST_DIRS_RE" || true)"
  if [ -n "$VIOL" ]; then
    echo "✗ 命中非白名单目录的外部 HTTP 调用："; echo "$VIOL"; FAIL=1
  fi
fi

echo "== [4/5] 配套 B9-6：模块禁自写 tenant_id / dataScope（收紧正则，避免误伤合法标识符）=="
# 仅匹配【手写赋值 / 比较 / 谓词构造】形态；\b 词边界 → roleDataScope / dataScopeValue / DataScopeOf 均不匹配
DENY_SCOPE='\btenant_id\s*[=:]|\bdataScope\s*(===|!==|==|!=)|\bswitch\s+[^;{]*\bdataScope\b|\bRowScope\s*\{|\bScope(All|Dept|IPGroup|Self)\b'
# 扫描 internal 全域后【排除白名单】：中间件 / db 层（含 scopedrepo.MustOwn）/ sqlc 生成 / SQL 查询
SCAN_INTERNAL='internal'
B9_WHITELIST_RE='^(internal/authmw/|internal/middleware/|internal/db/|db/query/)'
HITS_SCOPE="$(rg -l "$DENY_SCOPE" "$SCAN_INTERNAL" 2>/dev/null | grep -vE "$B9_WHITELIST_RE" || true)"
if [ -n "$HITS_SCOPE" ]; then
  echo "✗ 出现手写 tenant_id/dataScope（应经 authmw 注册表）："; echo "$HITS_SCOPE"; FAIL=1
fi

echo "== [5/5] 前端与依赖清单：禁 OPS SDK / 芋道 依赖（补齐前端覆盖）=="
# 前端源码/配置已在 [1]/[2] 扫描；此处补齐【依赖清单】检查（package.json + lockfile）
DENY_DEPS='"(ops|ops-sdk|ops-client|@ops/[a-z0-9._-]+|@shenyu/ops[a-z0-9._-]*)"|yudao|芋道'
if rg -n "$DENY_DEPS" -g 'package.json' -g 'package-lock.json' -g 'pnpm-lock.yaml' -g 'yarn.lock' . 2>/dev/null; then
  echo "✗ 前端依赖混入 OPS / 芋道 依赖"; FAIL=1
fi

if [ "$FAIL" -ne 0 ]; then
  echo "== §7A CI 门禁失败：禁止合并（禁止 --no-verify / 豁免注释绕过）=="
  exit 1
fi
echo "== §7A CI 门禁通过 =="
exit 0
```
